require "octokit"

class GithubCommitService
  def initialize(token:, repo_full_name:)
    # "github.com/org/repo" veya "org/repo" formatÄ±nÄ± temizle
    @repo = repo_full_name.to_s.sub(%r{\Ahttps?://github\.com/}, "").sub(%r{\Agithub\.com/}, "")
    @client = Octokit::Client.new(access_token: token)
  end

  def apply_hotfix(file_path:, new_content:, commit_message: "fix(guardrail): autonomous AI remediation hotfix")
    # 1. Deponun varsayÄ±lan dalÄ±nÄ± (default branch) bul
    repo_info = @client.repository(@repo)
    default_branch = repo_info.default_branch

    # 2. VarsayÄ±lan dalÄ±n son commit SHA'sÄ±nÄ± al
    base_ref = @client.ref(@repo, "heads/#{default_branch}")
    base_sha = base_ref.object.sha

    # 3. Hotfix iÃ§in yeni bir dal (branch) tÃ¼ret
    branch_name = "vaneai-hotfix-#{Time.now.to_i}"
    @client.create_ref(@repo, "heads/#{branch_name}", base_sha)

    # 4. Dosya zaten varsa SHA'sÄ±nÄ± al (update iÃ§in gerekir), yoksa yeni oluÅŸtur
    file_sha = nil
    begin
      existing_file = @client.contents(@repo, path: file_path, ref: branch_name)
      file_sha = existing_file.sha
    rescue Octokit::NotFound
      file_sha = nil
    end

    # 5. DosyayÄ± yeni dala commit'le
    commit_result = if file_sha
      @client.update_contents(@repo, file_path, commit_message, file_sha, new_content, branch: branch_name)
    else
      @client.create_contents(@repo, file_path, commit_message, new_content, branch: branch_name)
    end

    # 6. Otomatik bir Pull Request aÃ§
    pr = @client.create_pull_request(
      @repo,

import { NextRequest, NextResponse } from "next/server";

export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  const code = searchParams.get("code");

  if (!code) {
    return NextResponse.redirect(new URL("/?github_error=missing_code", request.url));
  }

  const clientId = process.env.NEXT_PUBLIC_GITHUB_CLIENT_ID;
  const clientSecret = process.env.GITHUB_CLIENT_SECRET;

  try {
    const tokenResponse = await fetch("https://github.com/login/oauth/access_token", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Accept: "application/json",
      },
      body: JSON.stringify({
        client_id: clientId,
        client_secret: clientSecret,
        code,
      }),
    });

    const tokenData = await tokenResponse.json();

    if (tokenData.error || !tokenData.access_token) {
      console.error("GitHub OAuth Error:", tokenData);
      return NextResponse.redirect(new URL("/?github_error=auth_failed", request.url));
    }

    const response = NextResponse.redirect(new URL("/?github_connected=true", request.url));

    // Token'Ä± 7 gÃ¼nlÃ¼k cookie olarak kaydediyoruz
    response.cookies.set("gh_token", tokenData.access_token, {
      httpOnly: false, // Ä°stemci tarafÄ±nda kontrol edilebilmesi iÃ§in
      secure: process.env.NODE_ENV === "production",
      path: "/",
      maxAge: 60 * 60 * 24 * 7,
    });

    return response;
  } catch (error) {
    console.error("GitHub Callback Exception:", error);

  try {
    const tokenRes = await fetch("https://github.com/login/oauth/access_token", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Accept: "application/json",
      },
      body: JSON.stringify({
        client_id: clientId,
        client_secret: clientSecret,
        code,
      }),
    });

    const tokenData = await tokenRes.json();

    if (tokenData.error || !tokenData.access_token) {
      console.error("GitHub OAuth error:", tokenData);
      return NextResponse.redirect(`${baseUrl}?github_error=${tokenData.error || "failed"}`);
    }

    const response = NextResponse.redirect(`${baseUrl}?github_connected=true`);
    response.cookies.set("gh_token", tokenData.access_token, {
      path: "/",
      httpOnly: false,
      secure: false,
      sameSite: "lax",
      maxAge: 60 * 60 * 24 * 7, // 7 gÃ¼n
    });

    return response;
  } catch (err) {
    console.error("OAuth exception:", err);
    return NextResponse.redirect(`${baseUrl}?github_error=internal_exception`);
  }
}

import { NextRequest, NextResponse } from "next/server";

export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  let repo = searchParams.get("repo");
  const path = searchParams.get("path") || "app/service.py";
  const token = request.cookies.get("gh_token")?.value;

  if (!token) {
    return NextResponse.json({ error: "Missing GitHub access token." }, { status: 401 });
  }

  if (!repo) {
    return NextResponse.json({ error: "Missing repo parameter." }, { status: 400 });
  }

  // URL ve Ã¶n ekleri temizle
  const cleanRepo = decodeURIComponent(repo)
    .replace(/^https?:\/\//, "")
    .replace(/^github\.com\//, "")
    .trim();

  try {
    const res = await fetch(`https://api.github.com/repos/${cleanRepo}/contents/${path}`, {
      headers: {
        Authorization: `Bearer ${token}`,
        Accept: "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
      },
      cache: "no-store",
    });

    if (!res.ok) {
      return NextResponse.json(
        { error: `File '${path}' could not be resolved in '${cleanRepo}'.` },
        { status: res.status }
      );
    }

    const data = await res.json();

    if (data.type !== "file" || !data.content) {
      return NextResponse.json(
        { error: `The path '${path}' does not point to a valid file.` },
        { status: 400 }
      );
    }

    const decodedContent = Buffer.from(data.content, "base64").toString("utf-8");

    return NextResponse.json({
      content: decodedContent,
      path: data.path,
      sha: data.sha,
    });
  } catch (error: any) {
    return NextResponse.json({ error: error.message || "Internal server error." }, { status: 500 });
  }
}
